package main

import (
	"fmt"
	"io"
	"net/http"
	"os"
)

func main() {
	body, err := os.Open("rex.png")
	if err != nil {
		panic(err)
	}
	defer body.Close()

	req, err := http.NewRequest("PUT", "https://api.example.com/pets/42/photo", body)
	if err != nil {
		panic(err)
	}
	req.Header.Set("Content-Type", "image/png")

	res, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer res.Body.Close()

	data, err := io.ReadAll(res.Body)
	if err != nil {
		panic(err)
	}
	fmt.Println(string(data))
}
