<?php

require __DIR__ . '/vendor/autoload.php';

use GuzzleHttp\Client;

$client = new Client();

$response = $client->request('POST', 'https://api.example.com/pets/42/photo', [
    'headers' => [
        'X-API-Key' => 'secret',
    ],
    'multipart' => [
        [
            'name' => 'caption',
            'contents' => 'Rex at the beach',
        ],
        [
            'name' => 'photo',
            'contents' => fopen('rex.png', 'r'),
            'filename' => 'rex.png',
            'headers' => [
                'Content-Type' => 'image/png',
            ],
        ],
    ],
]);

echo $response->getBody();
