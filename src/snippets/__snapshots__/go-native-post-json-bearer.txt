package main

import (
	"fmt"
	"io"
	"net/http"
	"strings"
)

func main() {
	body := strings.NewReader(`{
  "name": "Rex",
  "age": 3,
  "vaccinated": true,
  "owner": null,
  "tags": [
    "good",
    "boy"
  ],
  "address": {
    "city": "Berlin",
    "zip-code": "10115"
  },
  "empty": {},
  "list": []
}`)

	req, err := http.NewRequest("POST", "https://api.example.com/v1/pets", body)
	if err != nil {
		panic(err)
	}
	req.Header.Set("Authorization", "Bearer eyJhbGciOi.token")
	req.Header.Set("Content-Type", "application/json")

	res, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer res.Body.Close()

	data, err := io.ReadAll(res.Body)
	if err != nil {
		panic(err)
	}
	fmt.Println(string(data))
}
