request.setValue("it's \"quoted\" $HOME `tick` \\ ‘smart’", forHTTPHeaderField: "X-Nasty")

    it's "quoted" $HOME `tick` \\back\\slash
    new line	tab ünïcödé 😀 '@ \""" ${x}
    """.data(using: .utf8)
